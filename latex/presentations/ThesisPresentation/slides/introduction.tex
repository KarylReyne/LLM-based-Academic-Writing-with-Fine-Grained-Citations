\begin{frame}{Models}
    \begin{itemize}
        \item ScholarCopilot: \mono{ScholarCopilot-v1} \cite{wang2025scholarcopilot}
            \begin{itemize}
                \item backbone: \mono{Qwen2.5-7B} \cite{an2024qwen2}
                \item fine-tuned on the 500k ScholarCopilot training dataset
            \end{itemize}
        \item Passage Retriever: \mono{Qwen2.5-7B-Instruct} \cite{an2024qwen2}
        \item Generation Evaluation model ("judge"): \mono{DeepSeek-R1-Distill-Qwen-7B} \cite{guo2025deepseek}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Task 1: Reference Retrieval}
    Given a citing context, e.g. \cite{dosovitskiy2020image}:
    \begin{lstlisting}
        Transformers were proposed by Vaswani et al. (2017) for machine translation, and have since become the state of the art method in many NLP tasks. Large Transformer-based models are often pre-trained on large corpora and then fine-tuned for the task at hand: BERT (<|TARGET-CITATION|>
    \end{lstlisting}
    the model is tasked to retrieve the cited document.\\
    \bigskip
    Other citations -- here "Vaswani et al. (2017)" -- are either
    \begin{itemize}
        \item masked or
        \item replaced with the abstract of the cited document
    \end{itemize}
\end{frame}

\begin{frame}{Task 2: Generation Quality Evaluation}
    Given
    \begin{itemize}
        \item title
        \item abstract
        \item the first 2,5 sentences of the introduction
    \end{itemize}
    of an existing paper, the model is tasked to generate a scientific paper.\\
    \bigskip
    The judge evaluates the quality of the generated paper against the real paper by considering the following dimensions:
    \begin{itemize}
        \item Content Relevance
        \item Logical Coherence
        \item Academic Rigor
        \item Background Completeness
        \item Innovation Statement
    \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
    \begin{itemize}
        \item the PR module does \textbf{not improve ScholarCopilot} in either citation recall or generation quality
        \item the PR module is able to \textbf{identify correct citation targets}, even if ScholarCopilot fails to do so
        \item citations in experiments sections tend to be \textbf{easier to retrieve} compared to other sections
        \item LLM-based generation evaluation shows more consistent scoring behaviour when prompted to \textbf{provide reasoning} for its decision
    \end{itemize}
\end{frame}
