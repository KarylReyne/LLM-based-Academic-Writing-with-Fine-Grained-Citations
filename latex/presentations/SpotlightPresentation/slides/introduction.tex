\begin{frame}{Problem}
    Given a document $A$ (which may be generated) with a citation $c$, find the passage $p$ in the cited document $B$ that is most relevant to $A$.\\
    \bigskip
    If $A$ is generated, $p$ should ideally be the passage in $B$ that \emph{caused} the LLM to generate the statement that cites $B$ (this problem is introduced as \emph{context attribution} in \cite{cohenwang2024contextciteattributingmodelgeneration}). 
\end{frame}

\begin{frame}{Related Work}
    \textbf{ScholarCopilot} \cite{wang2025scholarcopilottraininglargelanguage}\\
    \begin{itemize}
        \item RAG with dynamic retrieval during generation via the generation of retrieval tokens
        \item focused on academic writing, currently introduction and related work sections 
    \end{itemize}
    \bigskip
    \textbf{ContextCite} \cite{cohenwang2024contextciteattributingmodelgeneration}\\
    \begin{itemize}
        \item method for scoring candidate context passages for \emph{contributive attribution} ('\emph{is this source the reason this statement was generated?}')
        \item determines how the prob. of the generated statement changes in a specific source is excluded
        \item evaluated on Wikipedia QA and news summarization datasets
    \end{itemize}
\end{frame}

\begin{frame}{Related Work, continued}
    \textbf{ReasonIR} \cite{shao2025reasonirtrainingretrieversreasoning}\\
    \begin{itemize}
        \item model+dataset for reasoning-intensive long context retrieval
        \item they also propose a method for generating synthetic training data
        \item dataset based on MS MARCO, Natural Questions + HotpotQA
    \end{itemize}
\end{frame}

\begin{frame}{Method - Dataset}
    \begin{itemize}
        \item citing document $A$: Fastformer \cite{wu2021fastformeradditiveattentionneed}
            \begin{itemize}
                \item citation context window size: 128 tokens
                \item context window centered at the citation
            \end{itemize}
        \item cited document $B$: Transformer \cite{vaswani2023attentionneed}
            \begin{itemize}
                \item sections / subsections split into chunks 
                \item chunk size: 512 tokens
            \end{itemize}
        \item this yields 4 citations (2x in introduction, 1x in methods and in experiments)
    \end{itemize}

\end{frame}

\begin{frame}{Method - Model}
    \textbf{Retriever: \mono{ReasonIR-8B}} \cite{shao2025reasonirtrainingretrieversreasoning}\\
    \begin{itemize}
        \item \mono{LLaMA3.1-8B} fine-tuned on the ReasonIR dataset
    \end{itemize}
    % \bigskip
    \textbf{Reranker: \mono{Qwen2.5-7B-Instruct}} \cite{qwen2, qwen2.5}\\
    \begin{itemize}
        \item additionally, the reranking score for each passage is averaged over $m$ LLM calls (this implements a simple version of batched self-consistency \cite{korikov2025batchedselfconsistencyimprovesllm}, see also \cite{wang2023selfconsistencyimproveschainthought})
        \item I currently use $m_{retrieval}=m_{reranking}=10$
    \end{itemize}
    % \bigskip
    \textbf{Final Score}\\
    \begin{itemize}
        \item interpolation of both ranking scores:\\
        $s_{final}=(1-\lambda)s_{retrieval}+\lambda s_{reranking}$ (I currently use $\lambda=0.5$)
    \end{itemize}
    \bigskip
    \textcolor{red}{All scores are currently obtained by comparing the query to every candidate document individually, this will be changed in the near future!}
\end{frame}
