\begin{frame}{Problem}
    Given a document $A$ (which may be generated) with a citation $c$, find the passage $p$ in the cited document $B$ that is most relevant to $A$.\\
    \bigskip
    If $A$ is generated, $p$ should ideally be the passage in $B$ that \emph{caused} the LLM to generate the statement that cites $B$. 
\end{frame}
